    1: import { useState, type ChangeEvent, type FormEvent } from 'react';
    2: import { z } from 'zod';
    3: import { MessageCircle } from 'lucide-react';
    4: import { useLocale } from '@/i18n/use-locale';
    5: import { trackWhatsAppClick } from '@/lib/whatsapp-tracking';
    6: 
    7: const initialValues = { model: '', year: '', mileage: '', issue: '', history: '', appointment: '', name: '', phone: '' };
    8: type Field = keyof typeof initialValues;
    9: 
   10: export default function MercedesServiceBookingForm() {
   11:   const { isArabic } = useLocale();
   12:   const [values, setValues] = useState(initialValues);
   13:   const [errors, setErrors] = useState<Partial<Record<Field, string>>>({});
   14:   const schema = z.object({
   15:     model: z.string().trim().min(2, isArabic ? 'يرجى إدخال الطراز' : 'Enter your Mercedes model').max(80),
   16:     year: z.string().trim().max(30),
   17:     mileage: z.string().trim().max(40),
   18:     issue: z.string().trim().min(2, isArabic ? 'أدخل رسالة الصيانة أو اكتب غير معروف' : 'Enter the service message, or write unknown').max(600),
   19:     history: z.string().trim().max(400),
   20:     appointment: z.string().trim().max(120),
   21:     name: z.string().trim().max(80),
   22:     phone: z.string().trim().max(20).refine(value => !value || (/^[+0-9\s()-]+$/.test(value) && value.replace(/\D/g, '').length >= 7), isArabic ? 'أدخل رقم هاتف صحيحاً أو اتركه فارغاً' : 'Enter a valid phone number or leave it blank'),
   23:   });
   24:   const fields: { key: Field; label: string; placeholder: string; limit: number; required?: boolean; multiline?: boolean }[] = isArabic ? [
   25:     { key: 'model', label: 'طراز مرسيدس', placeholder: 'مثلاً C200 أو GLC 300', limit: 80, required: true },
   26:     { key: 'year', label: 'سنة الصنع (إن عُرفت)', placeholder: 'مثلاً 2020', limit: 30 },
   27:     { key: 'mileage', label: 'المسافة المقطوعة والوحدة (إن عُرفت)', placeholder: 'مثلاً 45000 كم', limit: 40 },
   28:     { key: 'issue', label: 'رسالة Service A أو Service B أو رسالة الصيانة الأخرى', placeholder: 'انقل الرسالة، أو اكتب غير معروف. اذكر أي تحذير أو أعراض منفصلة.', limit: 600, required: true, multiline: true },
   29:     { key: 'history', label: 'سجل الصيانة السابق (إن توفر)', placeholder: 'تاريخ آخر خدمة والمسافة والأعمال المنفذة، أو غير معروف', limit: 400, multiline: true },
   30:     { key: 'appointment', label: 'الموعد المفضل (اختياري)', placeholder: 'اليوم والوقت المفضلان؛ يؤكد الفريق التوفر', limit: 120 },
   31:     { key: 'name', label: 'الاسم (اختياري)', placeholder: 'اسمك', limit: 80 },
   32:     { key: 'phone', label: 'رقم للاتصال (اختياري)', placeholder: '+971 50 000 0000', limit: 20 },
   33:   ] : [
   34:     { key: 'model', label: 'Mercedes model', placeholder: 'e.g. C200 or GLC 300', limit: 80, required: true },
   35:     { key: 'year', label: 'Model year (if known)', placeholder: 'e.g. 2020', limit: 30 },
   36:     { key: 'mileage', label: 'Mileage and unit (if known)', placeholder: 'e.g. 45,000 km', limit: 40 },
   37:     { key: 'issue', label: 'Service A/B reminder or other service message', placeholder: 'Copy the message, or write unknown. Mention any separate warning or symptoms.', limit: 600, required: true, multiline: true },
   38:     { key: 'history', label: 'Previous service history (if available)', placeholder: 'Last service date, mileage and completed work, or unknown', limit: 400, multiline: true },
   39:     { key: 'appointment', label: 'Preferred appointment (optional)', placeholder: 'Preferred day/time; the team confirms availability', limit: 120 },
   40:     { key: 'name', label: 'Name (optional)', placeholder: 'Your name', limit: 80 },
   41:     { key: 'phone', label: 'Callback number (optional)', placeholder: '+971 50 000 0000', limit: 20 },
   42:   ];
   43:   function submit(event: FormEvent) {
   44:     event.preventDefault();
   45:     const result = schema.safeParse(values);
   46:     if (!result.success) {
   47:       const nextErrors: Partial<Record<Field, string>> = {};
   48:       for (const issue of result.error.issues) nextErrors[issue.path[0] as Field] = issue.message;
   49:       setErrors(nextErrors);
   50:       return;
   51:     }
   52:     setErrors({});
   53:     const data = result.data;
   54:     const message = [
   55:       isArabic ? 'طلب موعد لصيانة مرسيدس من digitecme.com' : 'Mercedes service appointment request from digitecme.com',
   56:       ...fields.flatMap(field => data[field.key] ? [`${field.label}: ${data[field.key]}`] : []),
   57:     ].join('\n');
   58:     const url = `https://wa.me/97143402223?text=${encodeURIComponent(message)}`;
   59:     trackWhatsAppClick(url);
   60:     window.gtag?.('event', 'whatsapp_draft_opened', {
   61:       brand: 'Mercedes-Benz', page_path: window.location.pathname, form_id: 'brand-booking-form',
   62:       cta_placement: 'booking_form', service_intent: 'scheduled_maintenance',
   63:     });
   64:     window.open(url, '_blank', 'noopener,noreferrer');
   65:   }
   66:   const inputClass = 'w-full rounded-2xl border border-white/10 bg-black/60 px-4 py-3 text-off-white outline-none transition-colors placeholder:text-gray-500 focus:border-burnt-orange/60';
   67:   return (
   68:     <form id="brand-booking-form" data-mercedes-service-form onSubmit={submit} className="space-y-4" noValidate aria-describedby="mercedes-draft-note">
   69:       {fields.map(field => {
   70:         const common = {
   71:           id: `bf-${field.key}`, value: values[field.key], required: field.required,
   72:           onChange: (event: ChangeEvent<HTMLInputElement | HTMLTextAreaElement>) => setValues(previous => ({ ...previous, [field.key]: event.target.value })),
   73:           maxLength: field.limit, placeholder: field.placeholder, 'aria-invalid': Boolean(errors[field.key]),
   74:           'aria-describedby': errors[field.key] ? `bf-${field.key}-error` : undefined,
   75:         };
   76:         return <div key={field.key}>
   77:           <label htmlFor={common.id} className="mb-2 block text-sm text-gray-300">{field.label}</label>
   78:           {field.multiline ? <textarea {...common} rows={3} className={`${inputClass} resize-y`} /> : <input {...common} type={field.key === 'phone' ? 'tel' : 'text'} autoComplete={field.key === 'phone' ? 'tel' : field.key === 'name' ? 'name' : undefined} dir={field.key === 'phone' ? 'ltr' : undefined} className={inputClass} />}
   79:           {errors[field.key] && <p id={`bf-${field.key}-error`} role="alert" className="mt-1 text-xs text-burnt-orange">{errors[field.key]}</p>}
   80:         </div>;
   81:       })}
   82:       <button type="submit" className="inline-flex min-h-12 w-full items-center justify-center gap-2 rounded-2xl bg-burnt-orange px-6 py-4 font-bold text-black transition-colors hover:bg-burnt-orange/90 sm:w-auto">
   83:         <MessageCircle className="h-5 w-5 shrink-0" />{isArabic ? 'جهّز طلب الصيانة في واتساب' : 'Prepare service request in WhatsApp'}
   84:       </button>
   85:       <p id="mercedes-draft-note" className="text-xs leading-6 text-gray-400">{isArabic ? 'يكفي الطراز ورسالة الصيانة للبدء؛ يمكنك كتابة غير معروف. يجهز النموذج مسودة فقط. راجعها وأرسلها في واتساب ليتلقى الفريق الطلب. فتح المسودة لا يؤكد موعداً؛ يؤكد الفريق نطاق العمل والتوفر بعد الرد.' : 'Your model and service message are enough to start; you can write unknown. This form only prepares a draft. Review and send it in WhatsApp so the team receives your request. Opening the draft does not confirm an appointment; the team confirms the scope and availability when replying.'}</p>
   86:     </form>
   87:   );
   88: }
