    4: /** Keep route attribution without copying enquiry parameters or fragments. */
    5: export const analyticsPageUrl = (value: string, origin: string): string => {
    6:   try {
    7:     const url = new URL(value, origin);
    8:     return url.origin === origin ? `${url.origin}${url.pathname}` : origin;
    9:   } catch {
   10:     return origin;
   11:   }
   12: };
   13: 
   14: export const analyticsReferrer = (value: string): string => {
   15:   if (!value || value === 'direct') return 'direct';
   16:   try {
   17:     const url = new URL(value);
   18:     return ['http:', 'https:'].includes(url.protocol) ? url.origin : 'direct';
   19:   } catch {
   20:     return 'direct';
   21:   }
   22: };

   24: const sources = new Set(['google', 'bing', 'chatgpt', 'openai', 'perplexity', 'gemini', 'claude', 'copilot', 'duckduckgo', 'instagram', 'facebook', 'youtube', 'newsletter', 'whatsapp']);
   25: const media = new Set(['organic', 'cpc', 'ppc', 'paid_search', 'paid_social', 'social', 'email', 'referral', 'display', 'direct', 'organic_social']);
   26: const safeLabel = (value: unknown, allowed: Set<string>): string => {
   27:   const normalized = typeof value === 'string' ? value.toLowerCase().trim().replace(/^www\./, '').replace(/\.com$/, '') : '';
   28:   return allowed.has(normalized) ? normalized : '';
   29: };

   31: export function sanitizeAttribution(value: unknown, origin: string): ContactAttribution {
   32:   if (!value || typeof value !== 'object' || Array.isArray(value)) return {};
   33:   const input = value as Record<string, unknown>;
   34:   const output: ContactAttribution = {};
   35:   if (typeof input.landing_page === 'string') output.landing_page = analyticsPageUrl(input.landing_page, origin);
   36:   output.initial_referrer = analyticsReferrer(typeof input.initial_referrer === 'string' ? input.initial_referrer : '');
   37:   const source = safeLabel(input.utm_source, sources);
   38:   const medium = safeLabel(input.utm_medium, media);
   39:   if (source) output.utm_source = source;
   40:   if (medium) output.utm_medium = medium;
   41: 
   42:   const referrerHost = output.initial_referrer === 'direct' ? '' : new URL(output.initial_referrer).hostname;
   43:   const aiHosts: Record<string, string[]> = {
   44:     chatgpt: ['chatgpt.com', 'openai.com'],
   45:     perplexity: ['perplexity.ai'],
   46:     copilot: ['copilot.microsoft.com'],
   47:     gemini: ['gemini.google.com', 'bard.google.com'],
   48:     claude: ['claude.ai'],
   49:   };
   50:   for (const [name, hosts] of Object.entries(aiHosts)) {
   51:     if (source === name || (name === 'chatgpt' && source === 'openai') || hosts.some(host => referrerHost === host || referrerHost.endsWith(`.${host}`))) {
   52:       output.ai_referrer = name;
   53:       break;
   54:     }
   55:   }
   56:   // Arbitrary campaign/content/search terms and legacy stored keys can contain
   57:   // names or enquiry text. They are deliberately not copied into analytics.
   58:   return output;
   59: }

   61: export function buildAttribution(href: string, referrer: string): ContactAttribution {
   62:   const url = new URL(href);
   63:   return sanitizeAttribution({
   64:     landing_page: href,
   65:     initial_referrer: referrer,
   66:     utm_source: url.searchParams.get('utm_source'),
   67:     utm_medium: url.searchParams.get('utm_medium'),
   68:   }, url.origin);
   69: }
